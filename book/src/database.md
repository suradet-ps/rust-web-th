# เลเยอร์ฐานข้อมูล

ถ้าคุณเคยทำงานกับโปรเจกต์ที่ควิวรี SQL กระจัดกระจายอยู่ตามฟังก์ชันแฮนด์เลอร์ คุณจะรู้ว่ามันกลายเป็นความเจ็บปวดได้เร็วแค่ไหน ตอนแรกมันรู้สึกมีประสิทธิผล คุณกำลังส่งฟีเจอร์ออกไป สิ่งต่างๆ ทำงานได้ แต่ผ่านไปไม่กี่เดือน คุณต้องไล่หาข้ามไฟล์เป็นโหลเพื่อหาว่าต้องแก้ชื่อคอลัมน์ตรงไหน และเทสต์ของคุณก็ต้องพึ่งฐานข้อมูลที่รันอยู่สำหรับทุกอย่าง ผมเคยผ่านจุดนั้นมา และมันไม่สนุกเลย

ในบทนี้ เราจะสร้างเลเยอร์ฐานข้อมูลที่สะอาดและแยกส่วนอย่างดีด้วย SQLx ซึ่งเป็นตัวเลือกที่พบบ่อยที่สุดสำหรับแอปพลิเคชัน Axum อย่างไรก็ตาม หลักการเหล่านี้ใช้ได้ไม่ว่าคุณจะลงเอยด้วยครีตฐานข้อมูลตัวไหน

## การตั้งค่าพูลการเชื่อมต่อ

คุณไม่ต้องการให้ทุกคำขอที่เข้ามาเปิดการเชื่อมต่อฐานข้อมูลใหม่ ซึ่งหมายถึงการจับมือ TCP (TCP handshake) การเจรจา TLS และอาจรวมถึงการยืนยันตัวตน ทั้งหมดนี้ก่อนที่คุณจะได้รันควิวรีเสียอีก เราจึงรักษาพูลของการเชื่อมต่อที่สร้างไว้ล่วงหน้า ซึ่งคำขอต่างๆ สามารถยืมมาใช้แล้วส่งคืนได้

```rust
use anyhow::Context;
use sqlx::postgres::{PgPool, PgPoolOptions};
use std::time::Duration;

pub async fn create_pool(database_url: &str) -> anyhow::Result<PgPool> {
    let pool = PgPoolOptions::new()
        .max_connections(10)
        .min_connections(2)
        .acquire_timeout(Duration::from_secs(3))
        .idle_timeout(Duration::from_secs(600))
        .connect(database_url)
        .await
        .context("failed to connect to the database")?;

    Ok(pool)
}
```

ควรตั้งจำนวนการเชื่อมต่อเท่าไหร่? มันขึ้นอยู่กับระดับการทำงานพร้อมกันของคุณและสิ่งที่เซิร์ฟเวอร์ฐานข้อมูลรับไหว PostgreSQL มีค่าเริ่มต้นสูงสุด 100 การเชื่อมต่อ และแต่ละการเชื่อมต่อกินหน่วยความจำฝั่งเซิร์ฟเวอร์ จุดเริ่มต้นที่พอใช้ได้คือ 2 ถึง 4 การเชื่อมต่อต่อคอร์ CPU บนเซิร์ฟเวอร์แอปพลิเคชันของคุณ จากนั้นปรับตามเวลาที่คำขอของคุณใช้รอฐานข้อมูลจริงเทียบกับทำงานอย่างอื่นจริงๆ บนอินสแตนซ์ 4 คอร์ทั่วไป นั่นคือ 8 ถึง 16 การเชื่อมต่อ ถ้าคุณรันหลายอินสแตนซ์หลัง load balancer อย่าลืมว่าจำนวนการเชื่อมต่อรวมของทุกอินสแตนซ์ยังต้องต่ำกว่าขีดจำกัดของฐานข้อมูล

## ไมเกรชัน

ไมเกรชันของเราควรมีเวอร์ชัน ทำซ้ำได้ และในอุดมคติคือรันอัตโนมัติตอนเริ่มระบบ SQLx ให้ระบบไมเกรชันที่อ่านไฟล์ `.sql` จากไดเรกทอรีหนึ่งแล้วประยุกต์ใช้ตามลำดับ

```
migrations/
├── 20240101000000_create_users.sql
├── 20240102000000_create_posts.sql
└── 20240103000000_add_user_bio.sql
```

ไมเกรชันที่เขียนดีมีหน้าตาอย่างไร? ผมพยายามใส่ค่าเริ่มต้นที่สมเหตุสมผลสำหรับคอลัมน์ใหม่ ใช้ UUID แทนจำนวนเต็มที่เพิ่มอัตโนมัติสำหรับคีย์หลัก (ซึ่งทำให้การเดา ID ยากขึ้น ถึงแม้การอนุญาตสิทธิ์ต่างหากที่จะปกป้องแหล่งข้อมูลของคุณจริงๆ) และเก็บไทม์สแตมป์พร้อมข้อมูลเขตเวลา:

```sql
CREATE TABLE users (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email       VARCHAR(255) UNIQUE NOT NULL,
    name        VARCHAR(100) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    bio         TEXT,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_users_email ON users(email);

-- Automatically update the updated_at timestamp on row modification
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
```

ทำไมต้องใช้ทริกเกอร์ของฐานข้อมูลสำหรับ `updated_at`? เพราะมันทำให้สิ่งต่างๆ สอดคล้องกันไม่ว่าซอร์สโค้ดเส้นทางใดจะแก้ไขแถว แม้มีคนแก้ SQL ด้วยมือตรงๆ บนฐานข้อมูล (และเชื่อผมเถอะ มันเกิดขึ้นจริง) ไทม์สแตมป์ก็ยังถูกอัปเดตอย่างถูกต้อง

### การปั้นสกีมาอย่างประณีต

ไมเกรชันข้างต้นเป็นเพียงจุดเริ่มต้น ยังมีเทคนิคอีกสองสามอย่างที่เพิ่มตอนนี้ในไมเกรชันได้ง่าย แต่การนำมาเสริมทีหลังน่ารำคาญมาก ผมจะพาเดินดูเทคนิคที่ผมพบว่ามีประโยชน์ที่สุด

**ความเป็นเอกลักษณ์แบบไม่คำนึงตัวพิมพ์เล็กใหญ่ (Case-insensitive uniqueness)** ถ้าชื่อผู้ใช้และอีเมลควรเป็นเอกลักษณ์โดยไม่คำนึงถึงตัวพิมพ์เล็กใหญ่ ให้บังคับที่ระดับฐานข้อมูล เอ็กซ์เทนชัน `citext` ของ PostgreSQL หรือ collation ICU แบบ `COLLATE "und-ci-ai-ks-kc"` ป้องกันไม่ให้ "Alice" และ "alice" ถูกยอมรับทั้งคู่ ผมเคยเจอบั๊กจากการพึ่งการแปลงเป็นตัวพิมพ์เล็กในเลเยอร์แอปพลิเคชัน เพราะการ insert SQL โดยตรงหรือเซอร์วิสอื่นๆ สามารถเลี่ยงมันได้โดยสิ้นเชิง

**ตัวช่วยทริกเกอร์ที่ใช้ซ้ำได้** ถ้าตารางหลายตารางต้องการพฤติกรรม `updated_at` ให้แยกฟังก์ชันทริกเกอร์ออกมาใช้ครั้งเดียวแล้วใช้ซ้ำข้ามตาราง นั่นคือสิ่งที่ฟังก์ชัน `update_updated_at_column()` ข้างบนทำพอดี แค่ใช้ทริกเกอร์เดียวกันกับทุกตารางที่มีคอลัมน์ `updated_at`

**ชนิดอินเด็กซ์ที่เหมาะสม** สำหรับคอลัมน์อาร์เรย์อย่างเช่นแท็ก อินเด็กซ์ GIN เปิดใช้งานควิวรีแบบหาความครอบคลุมได้เร็ว (`@>`, `&&`) สำหรับคอลัมน์การค้นหาข้อความ ลองพิจารณา `tsvector` ร่วมกับอินเด็กซ์ GIN อินเด็กซ์ B-tree มาตรฐานเป็นตัวเลือกที่ถูกต้องสำหรับควิวรีแบบเท่ากับและแบบช่วงบนคอลัมน์สเกลาร์

**การเลือกชนิดข้อมูล ID อย่างจงใจ** UUID คือสิ่งที่ผมแนะนำเป็นค่าเริ่มต้นในคู่มือนี้ แต่มีบางสถานการณ์ที่ ID แบบเรียงลำดับ (bigserial) ทำงานได้ดีกว่า เช่น สเปกภายนอกที่คาดหวังจำนวนเต็ม หรือตารางที่มีการเขียนสูงซึ่งความสุ่มของ UUID ทำให้เกิดการแยกหน้า (page split) ของ B-tree UUIDv7 น่าคุยสำหรับตารางที่เน้นการเขียนซึ่งคุณต้องการทั้งความเป็นเอกลักษณ์และการเรียงลำดับที่เป็นมิตรกับอินเด็กซ์ เพราะมันฝังไทม์สแตมป์และเรียงตามลำดับเวลา PostgreSQL 18+ รองรับ `uuidv7()` แบบเนทีฟ สำหรับ PostgreSQL 17 และรุ่นก่อนหน้า คุณสร้างค่า UUIDv7 ในเลเยอร์แอปพลิเคชันได้โดยใช้ครีต `uuid` พร้อมฟีเจอร์ `v7` หรือใช้เอ็กซ์เทนชันของฐานข้อมูล

ถ้าคุณอยากเห็นแพตเทิร์นเหล่านี้ประยุกต์ใช้กับแอปพลิเคชันจริง รีโพสิทอรี [launchbadge/realworld-axum-sqlx](https://github.com/launchbadge/realworld-axum-sqlx) มีคำอธิบายไมเกรชันที่ยอดเยี่ยมคุ้มค่าแก่การอ่าน

### การรันไมเกรชัน

ผมชอบรันไมเกรชันตอนเริ่มแอปพลิเคชัน:

```rust
sqlx::migrate!("./migrations")
    .run(&pool)
    .await
    .context("failed to run database migrations")?;
```

แมโคร `migrate!` ฝังไฟล์ไมเกรชันเข้าไปในไบนารีของคุณตอนคอมไพล์ ดังนั้นคุณไม่ต้องส่งไดเรกทอรีไมเกรชันไปพร้อมกับแอปพลิเคชัน ลดเรื่องที่ต้องคิดระหว่างการดีพลอยต์ลงไปหนึ่งเรื่อง

### การตรวจสอบควิวรีตอนคอมไพล์ใน CI

มีเรื่องหนึ่งที่อาจทำให้คุณสะดุด: แมโคร `query!` และ `query_as!` ของ SQLx ตรวจสอบความถูกต้องของ SQL ของคุณกับฐานข้อมูลจริงตอนคอมไพล์ นั่นหมายความว่าสภาพแวดล้อม CI ของคุณต้องเข้าถึงฐานข้อมูลระหว่างบิลด์ มีสองวิธีในการจัดการเรื่องนี้

แนวทางที่ผมแนะนำสำหรับ CI คือ **โหมดออฟไลน์ (offline mode)** คุณรัน `cargo sqlx prepare` ในเครื่อง (ซึ่งเชื่อมต่อกับฐานข้อมูลที่รันอยู่และสร้างไดเรกทอรี `.sqlx/` ที่เก็บเมตาดาตาควิวรีแคช) จากนั้น commit ไดเรกทอรีนั้นเข้าระบบควบคุมเวอร์ชัน ใน CI ให้ตั้ง `SQLX_OFFLINE=true` แล้วแมโครจะใช้เมตาดาตาแคชแทนการเชื่อมต่อฐานข้อมูล เพิ่ม `cargo sqlx prepare --check` เป็นสเต็ปของ CI เพื่อตรวจสอบว่าเมตาดาตาแคชยังซิงก์กับควิวรีและไมเกรชันของคุณ

อีกทางเลือกหนึ่งคือปั่นฐานข้อมูลใน CI (ผ่าน Docker หรือเซอร์วิส CI) แล้วตั้ง `DATABASE_URL` เพื่อให้แมโครเชื่อมต่อได้โดยตรง การตั้งค่าง่ายกว่าแต่ช้ากว่าและเปราะบางกว่า

### หมายเหตุเรื่องไมเกรชันตอนเริ่มระบบ

การรันไมเกรชันตอนเริ่มระบบทำงานได้ดีสำหรับการดีพลอยต์แบบอินสแตนซ์เดียวและระหว่างการพัฒนา แต่ถ้าคุณรันหลายอินสแตนซ์ใน production คุณควรรู้ว่าทุกอินสแตนซ์จะพยายามรันไมเกรชันพร้อมกันตอนเริ่มระบบ SQLx จัดการเรื่องนี้ด้วย advisory lock ดังนั้นจริงๆ แล้วมีเพียงอินสแตนซ์เดียวเท่านั้นที่รันมัน แต่ในบางสถานการณ์ (โดยเฉพาะการดีพลอยต์แบบ rolling) งานไมเกรชันเฉพาะหรือ Kubernetes init container จะให้การควบคุมที่มากกว่าและการจัดการข้อผิดพลาดที่ชัดเจนกว่า เราจะเจาะลึกเรื่องนี้เพิ่มในบท [การดีพลอยต์](./deployment.md)

## แพตเทิร์นรีพอสิทอรี

นี่เป็นหนึ่งในแพตเทิร์นโปรดของผมสำหรับจัดการตรรกะฐานข้อมูลให้เป็นระเบียบ รีพอสิทอรีก็คือ struct ที่เป็นเจ้าของการเข้าถึงฐานข้อมูลทั้งหมดสำหรับแนวคิดโดเมนหนึ่งๆ มันถือการอ้างอิงถึงพูลการเชื่อมต่อ และเมธอดของมันจับคู่กับสิ่งที่คุณต้องทำจริงๆ ได้แก่ การสร้างเรคคอร์ด การค้นหาด้วยเกณฑ์ต่างๆ การอัปเดต และการลบ

```rust
#[derive(Clone)]
pub struct UserRepository {
    pool: PgPool,
}

impl UserRepository {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }

    pub async fn create(&self, req: &CreateUserRequest) -> Result<User, CreateUserError> {
        let row = sqlx::query_as!(
            UserRow,
            r#"
            INSERT INTO users (id, email, name, password_hash)
            VALUES ($1, $2, $3, $4)
            RETURNING id, email, name, password_hash, bio, created_at, updated_at
            "#,
            Uuid::new_v4(),
            req.email.as_str(),
            req.name.as_str(),
            &req.password_hash,
        )
        .fetch_one(&self.pool)
        .await
        .map_err(|e| match e {
            sqlx::Error::Database(ref db_err) if db_err.is_unique_violation() => {
                CreateUserError::Duplicate { email: req.email.clone() }
            }
            other => CreateUserError::Unknown(other.into()),
        })?;

        row.try_into()
            .map_err(|e: anyhow::Error| CreateUserError::Unknown(e))
    }

    pub async fn find_by_id(&self, id: &UserId) -> anyhow::Result<Option<User>> {
        let row = sqlx::query_as!(
            UserRow,
            r#"
            SELECT id, email, name, password_hash, bio, created_at, updated_at
            FROM users
            WHERE id = $1
            "#,
            id.as_uuid(),
        )
        .fetch_optional(&self.pool)
        .await
        .context("failed to fetch user by id")?;

        row.map(TryInto::try_into).transpose()
    }

    pub async fn find_by_email(&self, email: &Email) -> anyhow::Result<Option<User>> {
        let row = sqlx::query_as!(
            UserRow,
            r#"
            SELECT id, email, name, password_hash, bio, created_at, updated_at
            FROM users
            WHERE email = $1
            "#,
            email.as_str(),
        )
        .fetch_optional(&self.pool)
        .await
        .context("failed to fetch user by email")?;

        row.map(TryInto::try_into).transpose()
    }

    pub async fn update(
        &self,
        id: &UserId,
        name: Option<&UserName>,
        bio: Option<&str>,
    ) -> anyhow::Result<Option<User>> {
        let row = sqlx::query_as!(
            UserRow,
            r#"
            UPDATE users
            SET name = COALESCE($2, name),
                bio = COALESCE($3, bio)
            WHERE id = $1
            RETURNING id, email, name, password_hash, bio, created_at, updated_at
            "#,
            id.as_uuid(),
            name.map(|n| n.as_str()),
            bio,
        )
        .fetch_optional(&self.pool)
        .await
        .context("failed to update user")?;

        row.map(TryInto::try_into).transpose()
    }
}
```

ขอผมเน้นแพตเทิร์นไม่กี่อย่างที่น่าสังเกตตรงนี้

**ควิวรีที่ตรวจสอบตอนคอมไพล์** แมโคร `query_as!` ตรวจสอบความถูกต้องของ SQL ของคุณกับสกีมาฐานข้อมูลจริงตอนคอมไพล์และสร้างการแมปชนิดข้อมูลของตัวเองจากคอลัมน์ผลลัพธ์ควิวรีไปยังฟิลด์ของ struct มันไม่ใช้เทรต `FromRow` การแมปถูกสร้างโดยแมโครโดยตรงตามชื่อและชนิดข้อมูลของคอลัมน์ ถ้าคุณเปลี่ยนชื่อคอลัมน์ในไมเกรชันแต่ลืมอัปเดตควิวรี บิลด์จะล้มเหลว ผมพบว่านี่เป็นหนึ่งในฟีเจอร์ที่มีประโยชน์ที่สุดของ SQLx (ถ้าคุณชอบฟังก์ชัน `query_as::<_, UserRow>(sql)` แบบไม่มีแมโคร อันนั้นใช้ `FromRow` ที่รันไทม์)

**ชนิดข้อมูลแถวแบบแยกต่างหาก** struct `UserRow` จับคู่โดยตรงกับคอลัมน์ของฐานข้อมูล การ derive `sqlx::FromRow` ไม่จำเป็นสำหรับ `query_as!` แต่รวมไว้ก็ไม่เสียหายและมีประโยชน์ถ้าคุณใช้ฟังก์ชันควิวรีแบบไม่มีแมโครที่อื่นด้วย ชนิดข้อมูลโดเมน `User` มีนิวนิวไทป์สำหรับฟิลด์ของมันและอาจมีชื่อฟิลด์ที่ต่างกัน การแปลง `TryFrom<UserRow> for User` เป็นสะพานเชื่อมช่องว่างนี้ ดังที่เรากล่าวถึงในบท [การสร้างแบบจำลองโดเมน](./domain-modeling.md) เราใช้ `TryFrom` แทน `From` เพราะข้อมูลที่เก็บไว้อาจไม่เป็นไปตามอินแวเรียนต์ของโดเมนในปัจจุบัน

**`fetch_optional` สำหรับการค้นหา** เมื่อควิวรีด้วย ID หรืออีเมล ให้หยิบ `fetch_optional` แทน `fetch_one` เรคคอร์ดที่หายไปเป็นกรณีปกติ ไม่ใช่ข้อผิดพลาด และการคืน `Option<User>` เปิดโอกาสให้โค้ดผู้เรียกตัดสินใจว่า "ไม่พบ" เป็นปัญหาจริงๆ ในบริบทนั้นๆ หรือไม่

**COALESCE สำหรับการอัปเดตแบบบางส่วน** ฟังก์ชัน SQL `COALESCE` คงค่าเดิมไว้เมื่อพารามิเตอร์อัปเดตเป็น `NULL` ซึ่งให้คุณทำการอัปเดตบางส่วนได้โดยไม่ต้องอ่านค่าเดิมก่อน แต่มีข้อจำกัดที่ควรรู้: แพตเทิร์นนี้แยกไม่ออกระหว่าง "ผู้ใช้ส่ง null เพื่อล้างฟิลด์นี้" กับ "ผู้ใช้ไม่ได้รวมฟิลด์นี้ไว้" ทั้งสองกรณีมาถึงระดับ Rust เป็น `None` และทั้งคู่ถูกปฏิบัติเหมือน "คงค่าเดิมไว้" ถ้า API ของคุณต้องรองรับการล้างฟิลด์ที่เลือกได้อย่างชัดเจน คุณจะต้องใช้ค่า sentinel พารามิเตอร์ "ฟิลด์ที่ต้องการล้าง" แยกต่างหาก หรือการสร้าง SQL ที่เจาะจงชัดเจนกว่านี้

**ควิวรีแบบมีพารามิเตอร์ทุกที่** ทุกค่าที่ผู้ใช้ส่งเข้ามาผ่านตัวยึดตำแหน่งพารามิเตอร์ของ `.bind()` (`$1`, `$2`, ฯลฯ) ไม่เคยผ่านการ interpolate สตริง สิ่งนี้ป้องกันการแทรกแซง SQL (SQL injection) โดยการออกแบบ

## แพตเทิร์น FromRef เพื่อการเข้าถึงที่สะดวก

คุณอาจพบว่าตัวเองดึง `AppState` ทั้งหมดออกมาในทุกแฮนด์เลอร์เพียงเพื่อเข้าถึงพูลฐานข้อมูลหรือรีพอสิทอรี นั่นน่าเบื่อเร็วมาก แพตเทิร์น `FromRef` เสนอวิธีที่สะอาดกว่า โดยให้คุณดึงคอมโพเนนต์ย่อยของสถานะออกมาได้โดยตรง

```rust
use axum::extract::FromRef;

#[derive(Clone)]
pub struct AppState {
    pub db: PgPool,
    pub user_repo: UserRepository,
    // ... other fields
}

impl FromRef<AppState> for UserRepository {
    fn from_ref(state: &AppState) -> Self {
        state.user_repo.clone()
    }
}

impl FromRef<AppState> for PgPool {
    fn from_ref(state: &AppState) -> Self {
        state.db.clone()
    }
}
```

ตอนนี้แฮนด์เลอร์ของเราสามารถดึงแค่รีพอสิทอรีที่ต้องการ:

```rust
async fn get_user(
    State(repo): State<UserRepository>,
    Path(id): Path<Uuid>,
) -> AppResult<Json<UserResponse>> {
    let user = repo.find_by_id(&UserId::from_uuid(id))
        .await?
        .ok_or(AppError::NotFound)?;
    Ok(Json(user.into()))
}
```

ผมชอบสิ่งนี้เพราะมันทำให้ลายเซ็นของแฮนด์เลอร์โฟกัสกับสิ่งที่แฮนด์เลอร์ใช้จริง แทนที่จะดึงสถานะทั้งหมดของแอปพลิเคชันเข้ามาทุกครั้ง

## รีพอสิทอรีแบบเทรตเพื่อความสามารถในการทดสอบ

สำหรับกรณีที่ง่ายกว่า struct `UserRepository` แบบคอนกรีตที่เรียก SQLx ก็ทำงานได้ดีทีเดียว แต่ถ้าคุณต้องการเทสต์หน่วยเลเยอร์เซอร์วิสโดยไม่ต้องปั่นฐานข้อมูลขึ้นมา คุณสามารถนิยามรีพอสิทอรีเป็นเทรตแล้วให้ทั้งอิมพลีเมนเทชันจริงและอิมพลีเมนเทชันสำหรับเทสต์

เราจะกล่าวถึงเรื่องนี้โดยละเอียดในบท [แพตเทิร์นสถาปัตยกรรม](./architecture.md) ฉบับย่อคือ: นิยามเทรตในเลเยอร์โดเมนของคุณ implement ด้วย SQLx ในเลเยอร์อินฟราสตรัคเจอร์ แล้วใช้เจนเนอริกหรือเทรตออบเจกต์ในเลเยอร์เซอร์วิส จากนั้นเทสต์ของคุณสามารถให้อิมพลีเมนเทชันแบบในหน่วยความจำที่เก็บข้อมูลใน `HashMap` เท่านั้น

## ธุรกรรม

เมื่อคุณมีการเขียนฐานข้อมูลหลายรายการที่ต้องสำเร็จหรือล้มเหลวไปด้วยกัน ให้ห่อมันไว้ในธุรกรรม:

```rust
pub async fn transfer_credits(
    &self,
    from: &UserId,
    to: &UserId,
    amount: i64,
) -> anyhow::Result<()> {
    let mut tx = self.pool.begin().await.context("failed to begin transaction")?;

    sqlx::query!(
        "UPDATE accounts SET credits = credits - $1 WHERE user_id = $2",
        amount,
        from.as_uuid(),
    )
    .execute(&mut *tx)
    .await
    .context("failed to debit source account")?;

    sqlx::query!(
        "UPDATE accounts SET credits = credits + $1 WHERE user_id = $2",
        amount,
        to.as_uuid(),
    )
    .execute(&mut *tx)
    .await
    .context("failed to credit target account")?;

    tx.commit().await.context("failed to commit transfer")?;

    Ok(())
}
```

ถ้าขั้นตอนใดล้มเหลว (หรือฟังก์ชันคืนค่าเร็วกว่าปกติผ่าน `?`) ธุรกรรมจะถูก rollback โดยอัตโนมัติเมื่อตัวแปร `tx` ถูก drop ไม่ต้องใช้โค้ดทำความสะอาด

ถ้าคุณต้องการให้แฮนด์เลอร์ทั้งหมดรันภายในธุรกรรมเดียวที่ commit เมื่อสำเร็จและ rollback เมื่อเกิดข้อผิดพลาด ลองดูครีต `axum-sqlx-tx` มันให้ตัวแยกข้อมูล (extractor) ที่จัดการเรื่องนี้ให้คุณ ซึ่งสะดวกมากสำหรับธุรกรรมแบบจำกัดขอบเขตต่อคำขอ (request-scoped)

## การเลือกครีตฐานข้อมูล

คุณอาจสงสัยว่าควรใช้ครีตฐานข้อมูลตัวไหน นิเวศ Rust มีตัวเลือกที่แข็งแรงหลายตัว แต่ละตัวมีข้อแลกเปลี่ยนที่ต่างกัน

**SQLx** คือสิ่งที่คนส่วนใหญ่หยิบใช้กับ Axum มันเป็นแอซิงก์โดยธรรมชาติ รองรับการตรวจสอบควิวรีตอนคอมไพล์ และทำงานกับ SQL ดิบ คุณเขียน SQL เอง ซึ่งให้การควบคุมเต็มที่และหลีกเลี่ยงความไม่สอดคล้องเชิงอิมพีแดนซ์ (impedance mismatch) ที่ ORM บางครั้งนำเข้ามา การตรวจสอบตอนคอมไพล์จับความไม่ตรงกันของชนิดข้อมูลคอลัมน์ คอลัมน์ที่หายไป และข้อผิดพลาดทางไวยากรณ์ได้ก่อนที่โค้ดของคุณจะได้รัน

**Diesel** เป็น ORM ที่เก่าแก่และสมบูรณ์ที่สุดของ Rust มันให้ query builder ที่ปลอดภัยด้านชนิดข้อมูลซึ่งจับข้อผิดพลาดตอนคอมไพล์ผ่านระบบชนิดข้อมูลแทนการเชื่อมต่อฐานข้อมูล เดิม Diesel เป็นแบบซิงโครนัส แต่ครีต `diesel-async` ให้การรองรับแอซิงก์ มันเป็นตัวเลือกที่ดีถ้าคุณชอบ query builder มากกว่า SQL ดิบและต้องการการรับประกันที่แข็งแกร่งในตอนคอมไพล์

**SeaORM** ใช้แนวทางที่ได้แรงบันดาลใจจาก ActiveRecord พร้อมการรองรับแอซิงก์ในตัว มันสร้างชนิดข้อมูลเอนทิตีจากสกีมาฐานข้อมูลของคุณและให้ API ควิวรีระดับสูง มันเหมาะกับการพัฒนาอย่างรวดเร็วและโปรเจกต์ที่การเขียน SQL ดิบรู้สึกเกินจำเป็น ถึงแม้คุณจะได้การควบคุมแบบละเอียดน้อยกว่า SQLx

สำหรับโปรเจกต์ Axum ใหม่ส่วนใหญ่ ผมจะเลือก SQLx เป็นค่าเริ่มต้น มันเข้ากันโดยธรรมชาติกับปรัชญาของ Axum ที่เน้นความชัดเจนและการประกอบกัน (composable) มากกว่าความมหัศจรรย์ และการตรวจสอบควิวรีตอนคอมไพล์กลายเป็นการเพิ่มประสิทธิผลอย่างแท้จริงเมื่อคุณชินกับมันแล้ว
